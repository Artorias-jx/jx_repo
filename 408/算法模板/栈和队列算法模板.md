## 基础题

### 1. 判断字符串是否回文

**回文**：正读和反读都一样的字符串，如"abba"、"aba"

**方法1：使用栈**

**时间复杂度**：$O(n)$，空间复杂度：$O(n)$

```c
bool is_palindrome_stack(char str[], int n){
    char stack[MaxSize];
    int top = -1;
    
    // 将字符串前半部分入栈
    for (int i = 0; i < n / 2; i++){
        stack[++top] = str[i];
    }
    
    // 从中间位置开始比较
    int start = (n % 2 == 0) ? n / 2 : n / 2 + 1;
    
    for (int i = start; i < n; i++){
        if (stack[top--] != str[i])
            return false;
    }
    
    return true;
}
```

**方法2：双指针法（更高效）**

**时间复杂度**：$O(n)$，空间复杂度：$O(1)$

```c
bool is_palindrome_twopointer(char str[], int n){
    int left = 0, right = n - 1;
    
    while (left < right){
        if (str[left] != str[right])
            return false;
        left++;
        right--;
    }
    
    return true;
}
```

**要点**：
- 栈方法：前半部分入栈，后半部分与栈顶比较
- 双指针：从两端向中间比较，更节省空间
- 注意处理奇偶长度

### 2. 判断单链表中全部 n 个字符是否回文

**时间复杂度**：$O(n)$，空间复杂度：$O(n)$

```c
typedef struct CharNode{
    char data;
    struct CharNode *next;
} CharNode, *CharList;

bool is_palindrome_list(CharList L){
    if (L->next == NULL) return true;
    
    char stack[MaxSize];
    int top = -1;
    
    // 第一次遍历：统计长度，所有字符入栈
    CharNode *p = L->next;
    int n = 0;
    while (p != NULL){
        stack[++top] = p->data;
        p = p->next;
        n++;
    }
    
    // 第二次遍历：与栈中元素比较
    p = L->next;
    for (int i = 0; i < n; i++){
        if (p->data != stack[top--])
            return false;
        p = p->next;
    }
    
    return true;
}
```

**方法2：快慢指针 + 栈（优化空间）**

**空间复杂度**：$O(n/2)$

```c
bool is_palindrome_list_opt(CharList L){
    if (L->next == NULL) return true;
    
    // 快慢指针找中点
    CharNode *slow = L->next;
    CharNode *fast = L->next;
    char stack[MaxSize];
    int top = -1;
    
    // 前半部分入栈
    while (fast != NULL && fast->next != NULL){
        stack[++top] = slow->data;
        slow = slow->next;
        fast = fast->next->next;
    }
    
    // 奇数个结点，跳过中间结点
    if (fast != NULL)
        slow = slow->next;
    
    // 后半部分与栈比较
    while (slow != NULL){
        if (slow->data != stack[top--])
            return false;
        slow = slow->next;
    }
    
    return true;
}
```

**要点**：
- 方法1：所有字符入栈，再依次比较
- 方法2：用快慢指针找中点，只需前半部分入栈
- 链表无法随机访问，必须用栈或反转

### 3. 判断一个表达式中括号是否配对

**括号类型**：`()`, `[]`, `{}`

**时间复杂度**：$O(n)$，空间复杂度：$O(n)$

```c
bool is_valid_parentheses(char expr[], int n){
    char stack[MaxSize];
    int top = -1;
    
    for (int i = 0; i < n; i++){
        char c = expr[i];
        
        // 左括号入栈
        if (c == '(' || c == '[' || c == '{'){
            stack[++top] = c;
        }
        // 右括号匹配
        else if (c == ')' || c == ']' || c == '}'){
            if (top == -1)  // 栈空，无法匹配
                return false;
            
            char left = stack[top--];
            
            // 检查是否匹配
            if ((c == ')' && left != '(') ||
                (c == ']' && left != '[') ||
                (c == '}' && left != '{'))
                return false;
        }
    }
    
    // 栈应该为空
    return top == -1;
}
```

**要点**：
- 左括号入栈
- 遇到右括号，弹出栈顶检查是否匹配
- 最后栈应该为空
- 经典的栈应用

### 4. 判断子串 s2 是否匹配母串 s1，若匹配，输出匹配到第一个字符的索引

**KMP算法（模式匹配）**

**时间复杂度**：$O(n+m)$，空间复杂度：$O(m)$

```c
// 构建next数组（部分匹配表）
void get_next(char pattern[], int m, int next[]){
    next[0] = -1;
    int i = 0, j = -1;
    
    while (i < m - 1){
        if (j == -1 || pattern[i] == pattern[j]){
            i++;
            j++;
            next[i] = j;
        } else {
            j = next[j];
        }
    }
}

// KMP匹配
int kmp_match(char text[], int n, char pattern[], int m){
    int next[m];
    get_next(pattern, m, next);
    
    int i = 0, j = 0;
    
    while (i < n && j < m){
        if (j == -1 || text[i] == pattern[j]){
            i++;
            j++;
        } else {
            j = next[j];
        }
    }
    
    if (j == m)
        return i - m;  // 匹配成功，返回起始位置
    else
        return -1;     // 匹配失败
}
```

**朴素模式匹配（简单但效率低）**

**时间复杂度**：$O(n \cdot m)$

```c
int simple_match(char text[], int n, char pattern[], int m){
    for (int i = 0; i <= n - m; i++){
        int j;
        for (j = 0; j < m; j++){
            if (text[i + j] != pattern[j])
                break;
        }
        if (j == m)
            return i;  // 匹配成功
    }
    return -1;  // 匹配失败
}
```

**要点**：
- KMP算法：利用已匹配信息，避免重复比较
- next数组：记录模式串的最长相同前后缀
- 朴素算法：简单但效率低

### 5. 有两个链表 A 和 B，判断 B 是否是 A 的连续子序列

**时间复杂度**：$O(n \cdot m)$，空间复杂度：$O(1)$

```c
typedef struct LNode{
    int data;
    struct LNode *next;
} LNode, *LinkList;

bool is_subsequence(LinkList A, LinkList B){
    if (B->next == NULL) return true;  // 空链表B是任何链表的子序列
    if (A->next == NULL) return false;
    
    LNode *pa = A->next;
    
    // 遍历A，尝试每个位置开始匹配
    while (pa != NULL){
        LNode *p1 = pa;
        LNode *p2 = B->next;
        
        // 从当前位置开始匹配B
        while (p1 != NULL && p2 != NULL && p1->data == p2->data){
            p1 = p1->next;
            p2 = p2->next;
        }
        
        // B全部匹配完成
        if (p2 == NULL)
            return true;
        
        pa = pa->next;
    }
    
    return false;
}
```

**优化版本：KMP思想应用于链表**

```c
bool is_subsequence_kmp(LinkList A, LinkList B){
    if (B->next == NULL) return true;
    if (A->next == NULL) return false;
    
    LNode *pa = A->next;
    LNode *pb = B->next;
    LNode *start = pa;  // 记录匹配起始位置
    
    while (pa != NULL && pb != NULL){
        if (pa->data == pb->data){
            pa = pa->next;
            pb = pb->next;
        } else {
            // 匹配失败，从下一个位置重新开始
            start = start->next;
            pa = start;
            pb = B->next;
        }
    }
    
    return pb == NULL;
}
```

**要点**：
- 类似字符串匹配
- 从A的每个位置开始尝试匹配B
- 可以应用KMP思想优化

## 栈和队列的基本操作

### 顺序栈操作

```c
// 初始化
void init_stack(SqStack *S){
    S->top = -1;
}

// 判空
bool is_empty(SqStack S){
    return S.top == -1;
}

// 判满
bool is_full(SqStack S){
    return S.top == MaxSize - 1;
}

// 入栈
bool push(SqStack *S, int x){
    if (is_full(*S))
        return false;
    S->data[++S->top] = x;
    return true;
}

// 出栈
bool pop(SqStack *S, int *x){
    if (is_empty(*S))
        return false;
    *x = S->data[S->top--];
    return true;
}

// 获取栈顶元素
bool get_top(SqStack S, int *x){
    if (is_empty(S))
        return false;
    *x = S.data[S.top];
    return true;
}
```

### 循环队列操作

```c
// 初始化
void init_queue(SqQueue *Q){
    Q->front = Q->rear = 0;
}

// 判空
bool is_empty_queue(SqQueue Q){
    return Q.front == Q.rear;
}

// 判满
bool is_full_queue(SqQueue Q){
    return (Q.rear + 1) % MaxSize == Q.front;
}

// 入队
bool enqueue(SqQueue *Q, int x){
    if (is_full_queue(*Q))
        return false;
    Q->data[Q->rear] = x;
    Q->rear = (Q->rear + 1) % MaxSize;
    return true;
}

// 出队
bool dequeue(SqQueue *Q, int *x){
    if (is_empty_queue(*Q))
        return false;
    *x = Q->data[Q->front];
    Q->front = (Q->front + 1) % MaxSize;
    return true;
}
```

## 经典应用

### 表达式求值

**中缀表达式转后缀表达式（逆波兰表达式）**

```c
void infix_to_postfix(char infix[], char postfix[]){
    char stack[MaxSize];
    int top = -1;
    int k = 0;
    
    for (int i = 0; infix[i] != '\0'; i++){
        char c = infix[i];
        
        if (isdigit(c)){
            postfix[k++] = c;
        }
        else if (c == '('){
            stack[++top] = c;
        }
        else if (c == ')'){
            while (top != -1 && stack[top] != '(')
                postfix[k++] = stack[top--];
            top--;  // 弹出'('
        }
        else {  // 运算符
            while (top != -1 && stack[top] != '(' && 
                   priority(stack[top]) >= priority(c))
                postfix[k++] = stack[top--];
            stack[++top] = c;
        }
    }
    
    while (top != -1)
        postfix[k++] = stack[top--];
    
    postfix[k] = '\0';
}

int priority(char op){
    if (op == '+' || op == '-') return 1;
    if (op == '*' || op == '/') return 2;
    return 0;
}
```

**计算后缀表达式**

```c
int eval_postfix(char postfix[]){
    int stack[MaxSize];
    int top = -1;
    
    for (int i = 0; postfix[i] != '\0'; i++){
        char c = postfix[i];
        
        if (isdigit(c)){
            stack[++top] = c - '0';
        }
        else {
            int b = stack[top--];
            int a = stack[top--];
            
            switch(c){
                case '+': stack[++top] = a + b; break;
                case '-': stack[++top] = a - b; break;
                case '*': stack[++top] = a * b; break;
                case '/': stack[++top] = a / b; break;
            }
        }
    }
    
    return stack[top];
}
```

### 用两个栈实现队列

```c
typedef struct{
    SqStack s1, s2;
} MyQueue;

void init_myqueue(MyQueue *Q){
    init_stack(&Q->s1);
    init_stack(&Q->s2);
}

void enqueue_myqueue(MyQueue *Q, int x){
    push(&Q->s1, x);
}

bool dequeue_myqueue(MyQueue *Q, int *x){
    // 如果s2为空，将s1所有元素倒入s2
    if (is_empty(Q->s2)){
        while (!is_empty(Q->s1)){
            int temp;
            pop(&Q->s1, &temp);
            push(&Q->s2, temp);
        }
    }
    
    if (is_empty(Q->s2))
        return false;
    
    return pop(&Q->s2, x);
}
```

### 用两个队列实现栈

```c
typedef struct{
    SqQueue q1, q2;
} MyStack;

void init_mystack(MyStack *S){
    init_queue(&S->q1);
    init_queue(&S->q2);
}

void push_mystack(MyStack *S, int x){
    enqueue(&S->q1, x);
}

bool pop_mystack(MyStack *S, int *x){
    if (is_empty_queue(S->q1))
        return false;
    
    // 将q1中除最后一个元素外全部移到q2
    while (S->q1.front != (S->q1.rear - 1 + MaxSize) % MaxSize){
        int temp;
        dequeue(&S->q1, &temp);
        enqueue(&S->q2, temp);
    }
    
    // 弹出最后一个元素
    dequeue(&S->q1, x);
    
    // 交换q1和q2
    SqQueue temp = S->q1;
    S->q1 = S->q2;
    S->q2 = temp;
    
    return true;
}
```

## 总结

### 栈的应用

1. **括号匹配**：左括号入栈，右括号匹配
2. **表达式求值**：中缀转后缀，后缀求值
3. **函数调用**：函数调用栈
4. **回溯算法**：DFS、迷宫问题
5. **浏览器前进后退**：两个栈实现

### 队列的应用

1. **BFS**：图和树的层次遍历
2. **任务调度**：先来先服务
3. **缓冲区**：打印队列、消息队列
4. **循环队列**：解决假溢出问题

### 栈和队列的对比

| 特性 | 栈 | 队列 |
|------|-------|--------|
| 操作原则 | LIFO（后进先出） | FIFO（先进先出） |
| 操作 | push、pop | enqueue、dequeue |
| 应用 | 递归、回溯、括号匹配 | BFS、任务调度 |

### 时间复杂度

- **栈和队列的基本操作**：$O(1)$
- **回文判断**：$O(n)$
- **括号匹配**：$O(n)$
- **模式匹配（KMP）**：$O(n+m)$
