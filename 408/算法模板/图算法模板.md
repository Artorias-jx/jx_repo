## 数据结构定义

### 邻接矩阵

```c
#define MaxVertexNum 100

typedef struct{
    int Vex[MaxVertexNum];              // 顶点表
    int Edge[MaxVertexNum][MaxVertexNum]; // 邻接矩阵
    int vexnum, arcnum;                  // 顶点数和边数
} MGraph;
```

### 邻接表

```c
typedef struct ArcNode{
    int adjvex;              // 边指向的顶点位置
    struct ArcNode *next;    // 下一条边
    int weight;              // 边的权重（可选）
} ArcNode;

typedef struct VNode{
    int data;                // 顶点信息
    ArcNode *first;          // 第一条边
} VNode, AdjList[MaxVertexNum];

typedef struct{
    AdjList vertices;        // 邻接表
    int vexnum, arcnum;      // 顶点数和边数
} ALGraph;
```

## 基础题

### 1. 实现连通图 G（以邻接表存储）的广度优先遍历（BFS）

> [!note] ★★ 要会 —— 遍历序列是选择题常客；算法设计题**不考**「邻接表 + 指针」这种形态

**时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$

```c
void BFS_ALGraph(ALGraph G, int v){
    bool visited[MaxVertexNum] = {false};
    int queue[MaxVertexNum];
    int front = 0, rear = 0;
    
    printf("BFS遍历（邻接表）：");
    visited[v] = true;         // 入队时就标记已访问，防止同一顶点被重复入队
    printf("%d ", v);
    queue[rear++] = v;
    
    while (front < rear){
        int u = queue[front++];   // 出队一个顶点
        ArcNode *p = G.vertices[u].first;
        
        while (p != NULL){        // 沿 u 的边表扫描它的所有邻接点
            int w = p->adjvex;
            if (!visited[w]){
                visited[w] = true;
                printf("%d ", w);
                queue[rear++] = w;
            }
            p = p->next;
        }
    }
    printf("\n");
}
```

**要点**：
- 使用队列实现
- 从起点v开始，逐层访问邻接顶点
- visited数组防止重复访问

### 2. 实现连通图 G（以邻接矩阵存储）的广度优先遍历（BFS）

> [!note] ★★ 要会 —— 邻接矩阵正是 2021/2023/2024 三真题的存储形式，重点看

**时间复杂度**：$O(n^2)$，空间复杂度：$O(n)$

```c
void BFS_MGraph(MGraph G, int v){
    bool visited[MaxVertexNum] = {false};
    int queue[MaxVertexNum];
    int front = 0, rear = 0;
    
    printf("BFS遍历（邻接矩阵）：");
    visited[v] = true;
    printf("%d ", v);
    queue[rear++] = v;
    
    while (front < rear){
        int u = queue[front++];
        
        for (int w = 0; w < G.vexnum; w++){          // 邻接矩阵：要扫描整行
            if (G.Edge[u][w] != 0 && !visited[w]){   // 有边且未访问
                visited[w] = true;
                printf("%d ", w);
                queue[rear++] = w;
            }
        }
    }
    printf("\n");
}
```

**要点**：
- 邻接矩阵：需要遍历整行查找邻接顶点
- 时间复杂度较高：$O(n^2)$

### 3. 实现连通图 G（以邻接表存储）的深度优先遍历（DFS）

> [!note] ★★ 要会 —— 递归写法要能默写（选择题会问遍历序列与生成树）

**时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$

```c
void DFS_ALGraph(ALGraph G, int v, bool visited[]){
    visited[v] = true;   // 进入即标记
    printf("%d ", v);
    
    ArcNode *p = G.vertices[v].first;
    while (p != NULL){
        if (!visited[p->adjvex])          // 沿第一条未访问的边一路深入
            DFS_ALGraph(G, p->adjvex, visited);
        p = p->next;                      // 回溯后再看下一条边
    }
}

void DFS_traverse_AL(ALGraph G){
    bool visited[MaxVertexNum] = {false};
    printf("DFS遍历（邻接表）：");
    
    for (int i = 0; i < G.vexnum; i++){   // 外层循环：保证非连通图也能遍历到每个分量
        if (!visited[i])
            DFS_ALGraph(G, i, visited);
    }
    printf("\n");
}
```

**要点**：
- 递归实现
- 访问当前顶点后，递归访问未访问的邻接顶点

> **方法：BFS求最短路径，找距离最大的顶点**
>
> **时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$
>
> ```c
> int find_farthest(ALGraph G, int v){
>     int dist[MaxVertexNum];
>     bool visited[MaxVertexNum] = {false};
>     int queue[MaxVertexNum];
>     int front = 0, rear = 0;
>
>     // 初始化距离
>     for (int i = 0; i < G.vexnum; i++)

>
>         while (p != NULL){
>             int w = p->adjvex;
>             if (!visited[w]){

### 6. 求无向图的连通分量个数（DFS/BFS）

> [!note] ★★ 要会 —— 连通性选择题常客

**方法1：DFS**

**时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$

```c
void DFS_component(ALGraph G, int v, bool visited[]){
    visited[v] = true;
    ArcNode *p = G.vertices[v].first;
    
    while (p != NULL){
        if (!visited[p->adjvex])
            DFS_component(G, p->adjvex, visited);
        p = p->next;
    }
}

int count_components_DFS(ALGraph G){
    bool visited[MaxVertexNum] = {false};
    int count = 0;
    
    for (int i = 0; i < G.vexnum; i++){
        if (!visited[i]){           // 发现一个未访问顶点 → 一个新的连通分量
            DFS_component(G, i, visited);
            count++;                // 走完这一个分量，计数加 1
        }
    }
    
    return count;
}
```

**方法2：BFS**

```c
void BFS_component(ALGraph G, int v, bool visited[]){

    int front = 0, rear = 0;
    
    visited[v] = true;
    queue[rear++] = v;
    
    while (front < rear){
        int u = queue[front++];
        ArcNode *p = G.vertices[u].first;
        
        while (p != NULL){
            int w = p->adjvex;
            if (!visited[w]){
                visited[w] = true;
                queue[rear++] = w;
            }
            p = p->next;
        }
    }
}

int count_components_BFS(ALGraph G){
    bool visited[MaxVertexNum] = {false};
    int count = 0;
    
    for (int i = 0; i < G.vexnum; i++){
        if (!visited[i]){           // 未访问顶点 → 属于一个新的连通分量
            BFS_component(G, i, visited);
            count++;
        }
    }
    
    return count;
}
```

**要点**：
- 从每个未访问的顶点开始遍历
- 每次遍历完一个连通分量，计数加1

### 9. 有向图采用邻接表存储，计算每个顶点的入度和出度

> [!important] ★★★ 必背 —— **2023·41 K 顶点同型**（出度 > 入度）。只需会邻接矩阵版，见文末**骨架 B**

**时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$

```c
void calculate_degree(ALGraph G){
    int indegree[MaxVertexNum] = {0};
    int outdegree[MaxVertexNum] = {0};
    
    for (int i = 0; i < G.vexnum; i++){
        ArcNode *p = G.vertices[i].first;
        
        while (p != NULL){
            outdegree[i]++;          // i 发出一条边 → 出度 +1
            indegree[p->adjvex]++;   // 该边指向的顶点 → 入度 +1
            p = p->next;
        }
    }
    
    printf("顶点\t入度\t出度\n");
    for (int i = 0; i < G.vexnum; i++){
        printf("%d\t%d\t%d\n", i, indegree[i], outdegree[i]);
    }
}
```

**要点**：
- 出度：该顶点邻接表的边数
- 入度：其他顶点指向该顶点的边数

### 11. 设计算法判断无向图是否是一棵树

> [!note] ★★ 要会 —— 选择题常考「树 ⇔ 连通 + 边数 = 顶点数 − 1」

**判断条件**：连通 + 边数 = 顶点数 - 1

**时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$

```c
bool is_tree(ALGraph G){
    if (G.arcnum != G.vexnum - 1)   // 条件①：边数 = 顶点数 − 1
        return false;
    
    bool visited[MaxVertexNum] = {false};
    DFS_component(G, 0, visited);   // 条件②：连通性（从 0 出发能否走到所有顶点）
    
    for (int i = 0; i < G.vexnum; i++){
        if (!visited[i])
            return false;
    }
    
    return true;
}
```

**要点**：
- 树的充要条件：连通 + 边数 = 顶点数 - 1

### 13. 求不带权的无/有向图中 v 顶点到其他顶点的最短路径长度

> [!note] ★★ 要会 —— BFS 层次 = 无权最短路，遍历小题常客

**BFS求无权图的最短路径**

**时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$

```c
void bfs_shortest_path(ALGraph G, int v){
    int dist[MaxVertexNum];
    bool visited[MaxVertexNum] = {false};
    int queue[MaxVertexNum];
    int front = 0, rear = 0;
    
    for (int i = 0; i < G.vexnum; i++)
        dist[i] = -1;      // -1 表示不可达
    
    dist[v] = 0;           // 起点到自身距离为 0
    visited[v] = true;
    queue[rear++] = v;
    
    while (front < rear){
        int u = queue[front++];
        ArcNode *p = G.vertices[u].first;
        
        while (p != NULL){
            int w = p->adjvex;
            if (!visited[w]){
                visited[w] = true;
                dist[w] = dist[u] + 1;   // 邻接点距离 = 上一层 + 1（BFS 层数就是最短边数）
                queue[rear++] = w;
            }
            p = p->next;
        }
    }
    
    printf("从顶点 %d 到各顶点的最短路径长度：\n", v);
    for (int i = 0; i < G.vexnum; i++){
        printf("到顶点 %d：%d\n", i, dist[i]);
    }
}
```

**要点**：
- BFS保证先访问到的路径是最短的

### 14. Floyd 求有/无向图的最短路径长度

> [!note] ★★ 要会 —— **代码不必背**；要会手算与解释 $A^{(k)}$ 的含义（2015·42 考过 $A^2$）

**时间复杂度**：$O(n^3)$，空间复杂度：$O(n^2)$

```c
void floyd(MGraph G){
    int dist[MaxVertexNum][MaxVertexNum];
    
    for (int i = 0; i < G.vexnum; i++){          // 初始化：先放入直接边的权
        for (int j = 0; j < G.vexnum; j++){
            if (i == j)
                dist[i][j] = 0;
            else if (G.Edge[i][j] != 0)
                dist[i][j] = G.Edge[i][j];
            else
                dist[i][j] = 99999;              // 无边 → “无穷大”
        }
    }
    
    for (int k = 0; k < G.vexnum; k++){          // 逐个把顶点 k 当成“中转站”
        for (int i = 0; i < G.vexnum; i++){
            for (int j = 0; j < G.vexnum; j++){
                if (dist[i][k] + dist[k][j] < dist[i][j]){   // 经 k 中转更短 → 更新
                    dist[i][j] = dist[i][k] + dist[k][j];
                }
            }
        }
    }
    
    printf("最短路径矩阵：\n");
    for (int i = 0; i < G.vexnum; i++){
        for (int j = 0; j < G.vexnum; j++){
            printf("%d\t", dist[i][j]);
        }
        printf("\n");
    }
}
```

**要点**：
- 动态规划思想
- 枚举中间顶点k

### 15. 判断有向图是否有环（邻接表/邻接矩阵）

> [!important] ★★★ 必背 —— **2024·41「唯一拓扑序列」同族**；判环只是它的退化版（入队计数 < n），见文末**骨架 C**

**方法：拓扑排序**

**时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$

```c
bool has_cycle(ALGraph G){
    int indegree[MaxVertexNum] = {0};
    int queue[MaxVertexNum];
    int front = 0, rear = 0;
    int count = 0;
    
    for (int i = 0; i < G.vexnum; i++){    // 统计各顶点的入度
        ArcNode *p = G.vertices[i].first;
        while (p != NULL){
            indegree[p->adjvex]++;   // 每条边给“弧头”入度 +1
            p = p->next;
        }
    }
    
    for (int i = 0; i < G.vexnum; i++){
        if (indegree[i] == 0)        // 入度为 0 的顶点先入队
            queue[rear++] = i;
    }
    
    while (front < rear){
        int v = queue[front++];
        count++;                     // 记录已“输出”的顶点数
        
        ArcNode *p = G.vertices[v].first;
        while (p != NULL){
            indegree[p->adjvex]--;   // 删除 v 发出的边 → 邻接点入度 −1
            if (indegree[p->adjvex] == 0)
                queue[rear++] = p->adjvex;
            p = p->next;
        }
    }
    
    return count < G.vexnum;   // 没能输出全部顶点 → 存在环
}
```

**要点**：
- 如果能完成拓扑排序，则无环

### 16. 给定一个有向图 G (顶点编号 0...n)，边 (u→v) 表示任务 u 必须在任务 v 之前完成。请输出一组字典序最小的一种可执行顺序

> [!faq]- ☆ 冷门 —— **改用「数组选最小」代替最小堆**（PDF 官方解法，省掉 insert_heap/delete_min）
>
> **拓扑排序 + 每轮在零入度顶点里挑编号最小者**
>
> **时间复杂度**：$O(n^2)$（每轮线性找最小），空间复杂度：$O(n)$
>
> ```c
> bool topo_lexical(MGraph G, int order[]){
>     int indegree[MaxVertexNum] = {0};
>     int st[MaxVertexNum], top = -1, k = 0;
>     int n = G.numVertices;
>
>     for (int j = 0; j < n; j++)                  // 逐列统计入度
>         for (int i = 0; i < n; i++)
>             if (G.Edge[i][j] != 0) indegree[j]++;
>
>     for (int i = 0; i < n; i++)
>         if (indegree[i] == 0) st[++top] = i;     // 零入度顶点先入栈
>
>     while (top != -1){
>         int pos = 0;                             // 在栈内找编号最小的顶点
>         for (int p = 1; p <= top; p++)
>             if (st[p] < st[pos]) pos = p;
>         int v = st[pos];
>         st[pos] = st[top--];                     // 用栈顶覆盖，等价于删掉它
>         order[k++] = v;                          // 输出到 order 数组
>
>         for (int i = 0; i < n; i++)
>             if (G.Edge[v][i] != 0 && (--indegree[i] == 0))
>                 st[++top] = i;
>     }
>     return k == n;                               // 输出满 n 个 ⇒ 拓扑序存在
> }
> ```
>
> **要点**：
> - 每轮从零入度集合里取**编号最小**的顶点 → 全局字典序最小
> - 用普通数组 `st` + 线性找最小即可，**不需要最小堆**（这才是 408 该写的版本）
> - 返回 `k == n`：输出满 n 个顶点说明无环；否则有环、无拓扑序

### 17. 判断无向图是否有环

> [!faq]- ☆ 冷门 —— 未考，DFS 回边 / BFS 两种写法了解即可（PDF 两种都给）
>
> **方法一：DFS 判回边**
>
> **时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$
>
> ```c
> bool dfs_cycle(ALGraph G, int v, int parent, bool visited[]){
>     visited[v] = true;
>     ArcNode *p = G.vertices[v].first;
>
>     while (p != NULL){
>         int w = p->adjvex;
>
>         if (!visited[w]){
>             if (dfs_cycle(G, w, v, visited))
>                 return true;
>         } else if (w != parent){        // 撞上已访问且不是父亲 → 是回边 = 有环
>             return true;
>         }
>
>         p = p->next;
>     }
>
>     return false;
> }
>
> bool has_cycle_undirected(ALGraph G){
>     bool visited[MaxVertexNum] = {false};
>
>     for (int i = 0; i < G.vexnum; i++){
>         if (!visited[i]){
>             if (dfs_cycle(G, i, -1, visited))
>                 return true;
>         }
>     }
>
>     return false;
> }
> ```
>
> **方法二：BFS（用 `parent[]` 记录每个顶点的父结点）**
>
> **时间复杂度**：$O(n+e)$，空间复杂度：$O(n)$
>
> ```c
> bool has_cycle_bfs(ALGraph G){
>     bool visited[MaxVertexNum] = {false};
>     int parent[MaxVertexNum], queue[MaxVertexNum];
>     int front = 0, rear = 0;
>     for (int i = 0; i < G.vexnum; i++) parent[i] = -1;
>
>     visited[0] = true;
>     queue[rear++] = 0;
>
>     while (front < rear){
>         int v = queue[front++];
>         ArcNode *p = G.vertices[v].first;
>         while (p != NULL){
>             int w = p->adjvex;
>             if (!visited[w]){
>                 visited[w] = true;
>                 parent[w] = v;             // 记下 w 的父结点
>                 queue[rear++] = w;
>             } else if (w != parent[v]){    // 已访问且不是父亲 → 有环
>                 return true;
>             }
>             p = p->next;
>         }
>     }
>     return false;
> }
> ```
>
> **要点**：
> - 判据一致：**访问到已访问顶点且它不是父亲 ⇒ 有环**（无向图的一条边被"从两端各看一次"）
> - DFS 用递归 + `parent` 参数；BFS 用 `parent[]` 数组 + 队列
> - 上面只从 0 出发，非连通图要外层循环补跑每个未访问顶点
>

## 总结

### 图的遍历

| 方法 | 数据结构 | 应用 |
|------|----------|------|
| BFS | 队列 | 最短路径（无权图）、层次遍历 |
| DFS | 栈/递归 | 连通性、路径、环检测 |

### 最短路径算法

| 算法 | 适用 | 时间复杂度 |
|------|------|------------|
| BFS | 无权图 | $O(n+e)$ |
| Dijkstra | 非负权图，单源 | $O(n^2)$ 或 $O((n+e)\log n)$ |
| Floyd | 所有顶点对 | $O(n^3)$ |

### 拓扑排序

- **应用**：任务调度、依赖关系
- **判环**：拓扑排序失败 ↔ 有环
- **字典序最小**：用最小堆代替队列

## 近三年真题骨架（★★★ 必背）

> [!tip] 三题共用一个骨架
> **邻接矩阵 + 双层 `for` + 度数数组。** 一句话记牢：**行 → 出度，列 → 入度，整行非零个数 → 无向图的度。**

### 骨架 A · 2021·41 EL 路径（无向图，数奇度顶点个数）

**设计思想**：EL 路径存在 ⇔ 度为奇数的顶点个数为 0 或 2。所以只需统计奇度顶点数，不必真去求路径。

```c
int IsExistEL(MGraph G){
    int odd = 0;
    for (int i = 0; i < G.numVertices; i++){
        int deg = 0;
        for (int j = 0; j < G.numVertices; j++)
            if (G.Edge[i][j] != 0) deg++;      // 第 i 行非零元个数 = 顶点 i 的度
        if (deg % 2 != 0) odd++;
    }
    return (odd == 0 || odd == 2) ? 1 : 0;
}
```

时空：$O(n^2)$ / $O(1)$。

### 骨架 B · 2023·41 K 顶点（有向图，出度 > 入度）

```c
int printVertices(MGraph G){
    int cnt = 0;
    for (int i = 0; i < G.numVertices; i++){
        int out = 0, in = 0;
        for (int j = 0; j < G.numVertices; j++){
            if (G.Edge[i][j] != 0) out++;      // 第 i 行 → 出度
            if (G.Edge[j][i] != 0) in++;       // 第 i 列 → 入度
        }
        if (out > in){ printf("%c ", G.VerticesList[i]); cnt++; }
    }
    return cnt;
}
```

时空：$O(n^2)$ / $O(1)$。

### 骨架 C · 2024·41 唯一拓扑序列（每轮入度为 0 的顶点恰 1 个）

**设计思想**：拓扑排序每轮删掉一个入度为 0 的顶点。若**任何一轮**入度为 0 的顶点个数 ≠ 1，则拓扑序列不唯一（0 个 = 有环，≥2 个 = 有分支）。

```c
int uniquely(MGraph G){
    int indegree[MaxVertexNum] = {0};
    int n = G.numVertices;
    for (int j = 0; j < n; j++)                  // 逐列统计入度
        for (int i = 0; i < n; i++)
            if (G.Edge[i][j] != 0) indegree[j]++;

    for (int k = 0; k < n; k++){
        int zero = 0, v = -1;
        for (int i = 0; i < n; i++)
            if (indegree[i] == 0){ zero++; v = i; }
        if (zero != 1) return 0;                 // 0 个 → 有环；≥2 个 → 不唯一
        indegree[v] = -1;                        // -1 表示已输出，不再计入零度
        for (int j = 0; j < n; j++)
            if (G.Edge[v][j] != 0) indegree[j]--;
    }
    return 1;
}
```

时空：$O(n^2)$（邻接矩阵本身的遍历成本） / $O(n)$。
