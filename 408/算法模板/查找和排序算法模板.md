typedef struct BSTNode{
    int key;

## 一、查找算法

### 基础题

#### 1. 在有序数组中二分查找值为 key 的元素

**时间复杂度**：$O(\log n)$，空间复杂度：$O(1)$

```c
int binary_search(int arr[], int n, int key){
    int low = 0, high = n - 1;   // 在闭区间 [low, high] 内查找
    
    while (low <= high){                      // low > high 变成空区间 → 结束
        int mid = low + (high - low) / 2;     // 防溢出的 midpoint 写法，等价于 (low+high)/2
        
        if (arr[mid] == key)
            return mid;  // 找到，返回下标
        else if (arr[mid] < key)
            low = mid + 1;   // 目标在右半段
        else
            high = mid - 1;  // 目标在左半段
    }
    
    return -1;  // 未找到
}
```

**递归版本**：

```c
int binary_search_recursive(int arr[], int low, int high, int key){
    if (low > high)      // 出口：区间为空 → 没找到
        return -1;
    
    int mid = low + (high - low) / 2;
    
    if (arr[mid] == key)
        return mid;
    else if (arr[mid] < key)
        return binary_search_recursive(arr, mid + 1, high, key);   // 去右半段找
    else
        return binary_search_recursive(arr, low, mid - 1, key);    // 去左半段找
}
```

**要点**：
- 数组必须有序
- 每次比较后，查找范围缩小一半
- 注意边界条件：`low <= high`

#### 2. 寻找二叉排序树中最大值和最小值结点

**时间复杂度**：$O(h)$，空间复杂度：$O(1)$

```c
// 找最小值：一直向左
BSTNode* find_min(BSTree T){
    if (T == NULL) return NULL;
    
    while (T->lchild != NULL)   // 一路向左，最左端即最小值
        T = T->lchild;
    
    return T;
}

// 找最大值：一直向右
BSTNode* find_max(BSTree T){
    if (T == NULL) return NULL;
    
    while (T->rchild != NULL)   // 一路向右，最右端即最大值
        T = T->rchild;
    
    return T;
}
```

**递归版本**：

```c
BSTNode* find_min_recursive(BSTree T){
    if (T == NULL || T->lchild == NULL)   // 空 或 无左孩子 → 它就是最小
        return T;
    return find_min_recursive(T->lchild);
}

BSTNode* find_max_recursive(BSTree T){
    if (T == NULL || T->rchild == NULL)   // 空 或 无右孩子 → 它就是最大
        return T;
    return find_max_recursive(T->rchild);
}
```

**要点**：
- 最小值：最左边的结点
- 最大值：最右边的结点
- 利用BST的性质：左 < 根 < 右

#### 3. 求出值为 key 的结点在二叉排序树的层次

**时间复杂度**：$O(h)$，空间复杂度：$O(h)$

```c
int find_level_BST(BSTree T, int key, int level){
    if (T == NULL) return 0;
    
    if (T->key == key)
        return level;
    else if (key < T->key)
        return find_level_BST(T->lchild, key, level + 1);   // 比根小 → 只在左子树找
    else
        return find_level_BST(T->rchild, key, level + 1);   // 比根大 → 只在右子树找
}

// 调用方式
int get_level_BST(BSTree T, int key){
    return find_level_BST(T, key, 1);
}
```

**要点**：
- 利用BST的性质加速查找
- key < 根，向左查找
- key > 根，向右查找
- 层号从1开始

#### 4. 判断给定二叉树是否是二叉(搜索)排序树

**时间复杂度**：$O(n)$，空间复杂度：$O(h)$

```c
bool is_BST_helper(BSTree T, int min, int max){
    if (T == NULL) return true;
    
    // 当前结点的值必须严格落在 (min, max) 开区间内
    if (T->key <= min || T->key >= max)
        return false;
    
    // 递归检查左右子树（左子树的上界收紧为当前值，右子树的下界收紧为当前值）
    return is_BST_helper(T->lchild, min, T->key) && 
           is_BST_helper(T->rchild, T->key, max);
}

bool is_BST(BSTree T){
    return is_BST_helper(T, -99999, 99999);   // 初值相当于“负无穷 / 正无穷”
}
```

**方法2：中序遍历**

```c
bool is_BST_inorder(BSTree T, int *prev){
    if (T == NULL) return true;
    
    // 检查左子树
    if (!is_BST_inorder(T->lchild, prev))
        return false;
    
    // 检查当前结点：中序序列必须严格递增
    if (*prev != -99999 && T->key <= *prev)
        return false;
    *prev = T->key;   // 更新“上一个访问的值”
    
    // 检查右子树
    return is_BST_inorder(T->rchild, prev);
}

bool is_BST_v2(BSTree T){
    int prev = -99999;
    return is_BST_inorder(T, &prev);
}
```

**要点**：
- 方法1：递归检查每个结点的值是否在合法范围内
- 方法2：中序遍历BST得到递增序列，检查是否递增

#### 5. 查找二叉排序树中第 k 小的结点

**时间复杂度**：$O(n)$，空间复杂度：$O(h)$

```c
BSTNode* find_kth_smallest(BSTree T, int *k){
    if (T == NULL) return NULL;
    
    // 中序遍历：先走左
    BSTNode *left = find_kth_smallest(T->lchild, k);
    if (left != NULL) return left;   // 已在左子树中找到
    
    // 访问当前结点：计数减一，数到 0 即第 k 小
    (*k)--;
    if (*k == 0) return T;
    
    // 再走右
    return find_kth_smallest(T->rchild, k);
}

// 调用方式
BSTNode* find_kth(BSTree T, int k){
    int count = k;
    return find_kth_smallest(T, &count);
}
```

**要点**：
- 中序遍历BST得到有序序列
- 第k个访问的结点就是第k小的

#### 6. 求二叉搜索树中 p、q 两个结点的最近公共祖先

**时间复杂度**：$O(h)$，空间复杂度：$O(1)$

```c
BSTNode* find_lca_bst(BSTree T, int p, int q){
    if (T == NULL) return NULL;
    
    // p和q都在左子树 → 往左找
    if (p < T->key && q < T->key)
        return find_lca_bst(T->lchild, p, q);
    
    // p和q都在右子树 → 往右找
    if (p > T->key && q > T->key)
        return find_lca_bst(T->rchild, p, q);
    
    // 一大一小分居两侧（或其中一个是根）→ 当前结点即 LCA
    return T;
}
```

**要点**：
- 利用BST的性质
- 如果p和q分别在左右子树，当前结点就是LCA

#### 7. 二叉搜索树的最小绝对差

**时间复杂度**：$O(n)$，空间复杂度：$O(h)$

```c
void inorder_min_diff(BSTree T, int *prev, int *minDiff){
    if (T == NULL) return;
    
    inorder_min_diff(T->lchild, prev, minDiff);
    
    // 访问当前结点：与中序前驱做差（最小差一定出现在相邻两数之间）
    if (*prev != -1){
        int diff = T->key - *prev;
        if (diff < *minDiff)
            *minDiff = diff;
    }
    *prev = T->key;
    
    inorder_min_diff(T->rchild, prev, minDiff);
}

int min_absolute_diff(BSTree T){
    int prev = -1;
    int minDiff = 99999;
    inorder_min_diff(T, &prev, &minDiff);
    return minDiff;
}
```

**要点**：
- 中序遍历BST得到有序序列
- 最小差一定在相邻元素之间

#### 8. 判断一个二叉排序树是否为平衡二叉树

**时间复杂度**：$O(n)$，空间复杂度：$O(h)$

```c
int tree_height(BSTree T){
    if (T == NULL) return 0;
    int left = tree_height(T->lchild);
    int right = tree_height(T->rchild);
    return (left > right ? left : right) + 1;
}

bool is_balanced(BSTree T){
    if (T == NULL) return true;
    
    int left = tree_height(T->lchild);    // 求左右子树高度
    int right = tree_height(T->rchild);
    
    if (abs(left - right) > 1)   // 高度差 > 1 → 不平衡
        return false;
    
    return is_balanced(T->lchild) && is_balanced(T->rchild);   // 左右子树也要各自平衡
}
```

**优化版本（避免重复计算）**：

```c
int check_balanced(BSTree T, bool *balanced){
    if (T == NULL) return 0;
    
    int left = check_balanced(T->lchild, balanced);    // 后序：一次遍历同时求高度
    int right = check_balanced(T->rchild, balanced);
    
    if (abs(left - right) > 1)   // 顺手判平衡，省去重复求高度 → O(n) 而非 O(n²)
        *balanced = false;
    
    return (left > right ? left : right) + 1;   // 本子树高度回传给上层
}

bool is_balanced_opt(BSTree T){
    bool balanced = true;
    check_balanced(T, &balanced);
    return balanced;
}
```

**要点**：
- 平衡二叉树：任意结点的左右子树高度差 ≤ 1

## 二、排序算法

### 基础题

#### 1. 直接插入排序

**时间复杂度**：$O(n^2)$，空间复杂度：$O(1)$

```c
void insertion_sort(int arr[], int n){
    for (int i = 1; i < n; i++){   // 从第 2 个元素起，逐个插入前面的有序区
        int key = arr[i];          // 待插入元素
        int j = i - 1;
        
        // 将大于key的元素后移
        while (j >= 0 && arr[j] > key){
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;   // 空出来的位置放入 key
    }
}
```

**要点**：
- 从第二个元素开始，依次插入到前面的有序序列中
- 稳定排序
- 适合基本有序的数据

---

#### 2. 折半插入排序

**时间复杂度**：$O(n^2)$，空间复杂度：$O(1)$

```c
void binary_insertion_sort(int arr[], int n){
    for (int i = 1; i < n; i++){
        int key = arr[i];
        
        // 在有序区 [0, i-1] 内二分，找到 key 应插入的位置 left
        int left = 0, right = i - 1;
        while (left <= right){
            int mid = left + (right - left) / 2;
            if (arr[mid] > key)
                right = mid - 1;
            else
                left = mid + 1;
        }
        
        // 将元素后移（比较次数减少，但移动次数不变）
        for (int j = i - 1; j >= left; j--)
            arr[j + 1] = arr[j];
        
        arr[left] = key;
    }
}
```

**要点**：
- 用二分查找确定插入位置
- 减少比较次数，但移动次数不变
- 时间复杂度仍为 $O(n^2)$

#### 3. 选择排序

**时间复杂度**：$O(n^2)$，空间复杂度：$O(1)$

```c
void selection_sort(int arr[], int n){
    for (int i = 0; i < n - 1; i++){   // 每轮确定第 i 小的位置
        int min = i;                   // min 记录“未排序区最小元素”的下标
        
        // 找到最小元素的下标
        for (int j = i + 1; j < n; j++){
            if (arr[j] < arr[min])
                min = j;
        }
        
        // 交换
        if (min != i){
            int temp = arr[i];
            arr[i] = arr[min];
            arr[min] = temp;
        }
    }
}
```

**要点**：
- 每次从未排序部分选择最小元素
- 不稳定排序
- 比较次数固定：$\frac{n(n-1)}{2}$

---

#### 4. 冒泡排序

**时间复杂度**：$O(n^2)$，空间复杂度：$O(1)$

```c
void bubble_sort(int arr[], int n){
    for (int i = 0; i < n - 1; i++){
        bool swapped = false;
        
        for (int j = 0; j < n - i - 1; j++){   // 后 i 个元素已就位，不再比较
            if (arr[j] > arr[j + 1]){          // 相邻逆序则交换
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                swapped = true;
            }
        }
        
        // 优化：如果一趟没有交换，说明已有序
        if (!swapped) break;
    }
}
```

**要点**：
- 相邻元素两两比较，大的后移
- 稳定排序
- 优化：记录是否发生交换

#### 5. 快速排序

**时间复杂度**：平均 $O(n \log n)$，最坏 $O(n^2)$

**空间复杂度**：$O(\log n)$

```c
int partition(int arr[], int left, int right){
    int pivot = arr[left];          // 取最左元素作基准，先把它“挖出来”留下坑位
    int low = left, high = right;
    
    while (low < high){
        while (low < high && arr[high] >= pivot) high--;   // 从右找第一个 < pivot 的
        arr[low] = arr[high];                              // 填到左边的坑里
        while (low < high && arr[low] <= pivot) low++;     // 从左找第一个 > pivot 的
        arr[high] = arr[low];                              // 填到右边的坑里
    }
    arr[low] = pivot;   // 基准归位：左边都 ≤ 它，右边都 ≥ 它
    return low;         // 返回基准的最终下标
}

void quick_sort(int arr[], int left, int right){
    if (left >= right) return;   // 区间长度 ≤ 1，天然有序
    
    int pos = partition(arr, left, right);
    quick_sort(arr, left, pos - 1);    // 递归排左半段
    quick_sort(arr, pos + 1, right);   // 递归排右半段
}
```

**要点**：
- 分治思想：选择基准元素，分成两部分
- 不稳定排序
- 平均性能最好

#### 6. 二路归并排序

**时间复杂度**：$O(n \log n)$，空间复杂度：$O(n)$

```c
void merge(int arr[], int left, int mid, int right){
    int temp[right - left + 1];   // 辅助数组
    int i = left, j = mid + 1, k = 0;   // i 扫左半段，j 扫右半段
    
    // 合并两个有序数组
    while (i <= mid && j <= right){
        if (arr[i] <= arr[j])     // 用 <= 保证稳定性
            temp[k++] = arr[i++];
        else
            temp[k++] = arr[j++];
    }
    
    // 复制剩余元素
    while (i <= mid)
        temp[k++] = arr[i++];
    while (j <= right)
        temp[k++] = arr[j++];
    
    // 复制回原数组
    for (i = 0; i < k; i++)
        arr[left + i] = temp[i];
}

void merge_sort(int arr[], int left, int right){
    if (left >= right) return;
    
    int mid = left + (right - left) / 2;
    merge_sort(arr, left, mid);        // 递归分左半段
    merge_sort(arr, mid + 1, right);   // 递归分右半段
    merge(arr, left, mid, right);      // 合并两段有序序列
}
```

**要点**：
- 分治思想：分 → 治 → 合
- 稳定排序
- 适合链式存储

#### 7. 求数组 A 的第 k 小元素

**方法1：快速选择（基于快排的partition）**

**时间复杂度**：平均 $O(n)$，最坏 $O(n^2)$

```c
int quick_select(int arr[], int left, int right, int k){
    if (left >= right) return arr[left];
    
    int pos = partition(arr, left, right);   // 归位后 arr[pos] 就是第 pos+1 小的元素
    
    if (pos == k - 1)
        return arr[pos];                     // 恰好是第 k 小
    else if (pos > k - 1)
        return quick_select(arr, left, pos - 1, k);   // 第 k 小在左半段
    else
        return quick_select(arr, pos + 1, right, k);  // 第 k 小在右半段
}

// 调用方式
int find_kth_element(int arr[], int n, int k){
    return quick_select(arr, 0, n - 1, k);
}
```

**方法2：堆（维护大小为 k 的大根堆）**

**时间复杂度**：$O(n \log k)$

```c
// 大根堆下调（把 i 处的元素沉到合适位置）
void heapify_down_maxk(int heap[], int n, int i){
    int largest = i;
    int left = 2 * i + 1;    // 左孩子
    int right = 2 * i + 2;   // 右孩子
    
    if (left < n && heap[left] > heap[largest])
        largest = left;
    if (right < n && heap[right] > heap[largest])
        largest = right;
    
    if (largest != i){       // 孩子更大 → 交换后继续向下调整
        int temp = heap[i];
        heap[i] = heap[largest];
        heap[largest] = temp;
        heapify_down_maxk(heap, n, largest);
    }
}

int find_kth_heap(int arr[], int n, int k){
    int heap[k];
    
    // 前 k 个元素建大根堆（堆顶 = 当前 k 个中的最大值 = 第 k 小）
    for (int i = 0; i < k; i++)
        heap[i] = arr[i];
    
    for (int i = k/2 - 1; i >= 0; i--)   // 从最后一个非叶结点起，自下而上调整成堆
        heapify_down_maxk(heap, k, i);
    
    // 处理剩余元素：小于堆顶则替换（维护最小的 k 个）
    for (int i = k; i < n; i++){
        if (arr[i] < heap[0]){      // 注意：比堆顶小才入堆
            heap[0] = arr[i];
            heapify_down_maxk(heap, k, 0);
        }
    }
    
    return heap[0];  // 大根堆堆顶 = 第 k 小的元素
}
```

**要点**：
- 快速选择：平均 $O(n)$，不需要完全排序
- 堆方法：适合海量数据
- Top K 问题的经典解法

## 三、其他常用排序

### 希尔排序

**时间复杂度**：$O(n^{1.3})$ ~ $O(n^2)$

```c
void shell_sort(int arr[], int n){
    for (int gap = n/2; gap > 0; gap /= 2){   // 增量逐步缩小，最后 gap=1
        for (int i = gap; i < n; i++){        // 对每个“间隔 gap 的子序列”做插入排序
            int temp = arr[i];
            int j;
            for (j = i; j >= gap && arr[j - gap] > temp; j -= gap)
                arr[j] = arr[j - gap];        // 组内元素后移 gap 位
            arr[j] = temp;
        }
    }
}
```

### 堆排序

**时间复杂度**：$O(n \log n)$，空间复杂度：$O(1)$

```c
void heapify_down_max(int arr[], int n, int i){
    int largest = i;
    int left = 2 * i + 1;    // 左孩子下标
    int right = 2 * i + 2;   // 右孩子下标
    
    if (left < n && arr[left] > arr[largest])
        largest = left;
    if (right < n && arr[right] > arr[largest])
        largest = right;
    
    if (largest != i){       // 孩子更大 → 交换后继续下沉
        int temp = arr[i];
        arr[i] = arr[largest];
        arr[largest] = temp;
        heapify_down_max(arr, n, largest);
    }
}

void heap_sort(int arr[], int n){
    // 建大根堆
    for (int i = n/2 - 1; i >= 0; i--)   // 从最后一个非叶结点起自下而上调整
        heapify_down_max(arr, n, i);
    
    // 排序
    for (int i = n - 1; i > 0; i--){
        int temp = arr[0];    // 堆顶（当前最大）换到末尾
        arr[0] = arr[i];
        arr[i] = temp;
        heapify_down_max(arr, i, 0);   // 剩下 i 个元素重新调整堆
    }
}
```

### 计数排序

**时间复杂度**：$O(n+k)$，$k$为数据范围

```c
void counting_sort(int arr[], int n){
    int max = arr[0], min = arr[0];
    
    for (int i = 1; i < n; i++){    // 先确定数据范围
        if (arr[i] > max) max = arr[i];
        if (arr[i] < min) min = arr[i];
    }
    
    int range = max - min + 1;
    int count[range];
    int output[n];
    
    for (int i = 0; i < range; i++)   // 计数数组清零
        count[i] = 0;
    
    for (int i = 0; i < n; i++)       // 统计每个值出现次数（用 arr[i]-min 作下标，可兼容负数）
        count[arr[i] - min]++;
    
    for (int i = 1; i < range; i++)   // 前缀和 → count[v] 变为“值 ≤ v 的元素个数”
        count[i] += count[i - 1];
    
    for (int i = n - 1; i >= 0; i--){   // 从后往前放 → 保持稳定
        output[count[arr[i] - min] - 1] = arr[i];
        count[arr[i] - min]--;
    }
    
    for (int i = 0; i < n; i++)   // 写回原数组
        arr[i] = output[i];
}
```

## 总结

### 查找算法对比

| 算法 | 时间复杂度（平均） | 空间复杂度 | 适用场景 |
|------|-------------------|-----------|----------|
| 顺序查找 | $O(n)$ | $O(1)$ | 无序表 |
| 二分查找 | $O(\log n)$ | $O(1)$ | 有序表 |
| BST查找 | $O(\log n)$ | $O(h)$ | 动态查找 |
| 哈希查找 | $O(1)$ | $O(n)$ | 快速查找 |

### 排序算法对比

| 算法 | 时间复杂度（平均） | 时间复杂度（最坏） | 空间复杂度 | 稳定性 |
|------|-------------------|-------------------|-----------|--------|
| 冒泡排序 | $O(n^2)$ | $O(n^2)$ | $O(1)$ | 稳定 |
| 选择排序 | $O(n^2)$ | $O(n^2)$ | $O(1)$ | 不稳定 |
| 插入排序 | $O(n^2)$ | $O(n^2)$ | $O(1)$ | 稳定 |
| 希尔排序 | $O(n^{1.3})$ | $O(n^2)$ | $O(1)$ | 不稳定 |
| 归并排序 | $O(n \log n)$ | $O(n \log n)$ | $O(n)$ | 稳定 |
| 快速排序 | $O(n \log n)$ | $O(n^2)$ | $O(\log n)$ | 不稳定 |
| 堆排序 | $O(n \log n)$ | $O(n \log n)$ | $O(1)$ | 不稳定 |
| 计数排序 | $O(n+k)$ | $O(n+k)$ | $O(k)$ | 稳定 |

### 排序算法选择

- **数据量小**（n < 50）：直接插入排序
- **数据量中等**：快速排序
- **数据量大且要求稳定**：归并排序
- **数据基本有序**：直接插入排序、冒泡排序
- **数据范围小**：计数排序
- **不关心平均性能，只要最坏情况好**：堆排序

### Top K 问题

- **方法1**：快速选择，平均 $O(n)$
- **方法2**：维护大小为k的堆，$O(n \log k)$
- **方法3**：完全排序，$O(n \log n)$

选择：数据量不大用方法1，海量数据用方法2
